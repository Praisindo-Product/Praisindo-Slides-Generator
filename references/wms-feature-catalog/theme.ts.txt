import { Font, StyleSheet } from "@react-pdf/renderer";
import path from "node:path";

/**
 * The catalog is drawn in the product's own document language, the one the Consolidated Client
 * Statement uses (src/features/wealth/profiles/statement-pdf/statement-document.tsx): Libre
 * Franklin, the light theme's tokens, a brand header and a hairline footer on every page, bordered
 * cards and tables with a raised header row, figure strips and tinted tags. Keep the two in step.
 */

export const ROOT = process.cwd().replace(/\\/g, "/");
export const FONT = "Libre Franklin";

let registered = false;
export function registerFonts(): void {
  if (registered) return;
  registered = true;
  Font.register({
    family: FONT,
    fonts: [400, 500, 600, 700].map((fontWeight) => ({ src: path.join(ROOT, "public/fonts/statement/libre-franklin-" + fontWeight + ".ttf"), fontWeight })),
  });
  // A module or capability name split with a hyphen reads as a typo, so nothing hyphenates.
  Font.registerHyphenationCallback((word) => [word]);
}
